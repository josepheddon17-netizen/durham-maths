
\section{Parametrisation and Curves}

\begin{definition}[Curve]
 A curve $C$ is a 1-d subset of $\mathbb{R}^n$, for $n>1$. 
\end{definition}

\noindent It will be more use to us to define a curve through parametrisation.

\begin{definition}[Parametrisation]
  \emph{Parametrisation} is a function, $\boldsymbol{x}(t)$, which maps a real interval $\left[t_0, t_1\right]$ into $\mathbb{R}^n$. This is a vector-valued function

$$
\boldsymbol{x}(t)=x_1(t) \boldsymbol{e}_1+x_2(t) \boldsymbol{e}_2+\ldots+x_n(t) \boldsymbol{e}_n,
$$

where $\left\{\boldsymbol{e}_1, \boldsymbol{e}_2, \ldots, \boldsymbol{e}_n\right\}$ denote the Cartesian unit vectors. When $\boldsymbol{n}=\mathbf{2}$ or $\boldsymbol{n}=\mathbf{3}$, we will usually write $x_1 \equiv x, x_2 \equiv y$ and $x_3 \equiv z$.
\end{definition}

\begin{definition}[Closed and Simple Curve]
  
If $\boldsymbol{x}\left(t_1\right)=\boldsymbol{x}\left(t_0\right)$ then the curve is called closed. A curve that does not cross itself is called simple.

\end{definition}

\begin{remark}
  A common parametrisation for a circle is polar coordinates. We note here however that parametrisation is not unique and a single curve may be described by multiple parametrisation.
\end{remark}

\begin{definition}[Differentaible Parametrisation]
  A parametrisation $\boldsymbol{x}(t)$ is differentiable if it has a well-defined tangent vector,

$$
\frac{d \boldsymbol{x}}{d t}=\lim _{d t \rightarrow 0} \frac{\boldsymbol{x}(t+d t)-\boldsymbol{x}(t)}{d t},
$$

so-called because its direction points along the curve at each $t$.
\end{definition}

\begin{ndefi}[Regular Parametrisation]
	A parametrisation is regular if $\frac{d \boldsymbol{x}}{d t} \neq 0$ everywhere, and a curve is regular if it has at least one regular parametrisation.
\end{ndefi}

\begin{ndefi}[Singular Parametrisation and Cusps]
	
	A parametrisation is singular if $\frac{d \boldsymbol{x}}{d t} = 0$ at at least one point. The singular points of a parametrisation are called susps.
	
\end{ndefi}

\begin{nrem}
	You can construct a singular parametrisation even for a regular curve.
\end{nrem}


\begin{remark}
  If we think of $\boldsymbol{x}(t)$ as the trajectory of a particle, then $\left|\frac{d \boldsymbol{x}}{d t}\right|$ is the speed at which the particle is moving, for each $t$.
\end{remark}

\begin{theorem}[Chain rule for parametrisation]
 If $\boldsymbol{x}(t)$ and $\boldsymbol{x}(\tau)$ are two different parametrisations of the same curve, then the chain rule gives
$$
\frac{d \boldsymbol{x}}{d t}=\frac{d \boldsymbol{x}}{d \tau} \frac{d \tau}{d t} .
$$ 
\end{theorem}

\section{Line Integral of Scalar Fields}

\begin{definition}[Line Integral of Scalar Field]
  The line integral of a scalar field $f(\boldsymbol{x})$ along a curve $C$ with parametrisation $\boldsymbol{x}(t)$, for $t \in\left[t_0, t_1\right]$, is defined as

$$
\int_C f d \ell=\int_{t_0}^{t_1} f(\boldsymbol{x}(t))\left|\frac{d \boldsymbol{x}}{d t}\right| d t
$$
\end{definition}

\begin{nprop}[Formula for Scalar Line Element]
	$\mathrm{d} \ell = \pm\left|\mathbf{x}^{\prime}(t)\right| \mathrm{d} t$
\end{nprop}

\begin{definition}[Rectifiable]
  
A curve which has finite length is called rectifiable. This is a weaker condition than "regular" (for example, if $\boldsymbol{x}(t)$ is Lipschitz continuous then it is rectifiable).

\end{definition}


\begin{nrem}[Arclength]
	We can derive the arc length between two points on the curve $\mathbf{r}(u)$, given by $u=u_1$ and $u=u_2$, as
	
	\begin{equation}
		s=\int_{u_1}^{u_2} \sqrt{\frac{d \mathbf{r}}{d u} \cdot \frac{d \mathbf{r}}{d u}} d u 
	\end{equation}
	
\end{nrem}

\begin{proof}
  The tiny displacement vector is
	
	$$
	d \mathbf{r}=\mathbf{r}^{\prime}(u) d u
	$$
	
	
	Its magnitude gives the tiny distance travelled:
	
	$$
	d s=|d \mathbf{r}|
	$$
	
	
	Hence
	
	$$
	d s=\left|\mathbf{r}^{\prime}(u) d u\right|
	$$
	
	and since $d u$ is scalar,
	
	$$
	d s=\left|\mathbf{r}^{\prime}(u)\right||d u|
	$$
	
	which becomes
	
	$$
	d s=\left|\mathbf{r}^{\prime}(u)\right| d u
	$$

  This can be directly integrated to find our arclength formula.
\end{proof}


\section{Line Integrals of Vector Fields}

\begin{definition}[Unit Tangent Vector of Parametrisation]
  The unit tangent vector of a parametrisation is

$$
\hat{\boldsymbol{t}}=\frac{\frac{d \boldsymbol{x}}{d t}}{\left|\frac{d \boldsymbol{x}}{d t}\right|} .
$$

This only depends on the direction of the parametrisation, not the "speed".
\end{definition}

\begin{definition}[Oriented Curve]
  An oriented curve is a curve together with a specified (consistent) choice of $\hat{\mathbf{t}}$
\end{definition}

\begin{definition}[Line Integral of a Vector Field along Oriented Curve]
  The line integral of a vector field $\boldsymbol{f}(\boldsymbol{x})$ along an oriented curve $C$ with parametrisation $\boldsymbol{x}(t)$, for $t \in\left[t_0, t_1\right]$, is defined as the scalar line integral of the tangential component:

$$
\int_C \boldsymbol{f} \cdot \hat{\boldsymbol{t}} d \ell=\int_{t_0}^{t_1} \boldsymbol{f}(\boldsymbol{x}(t)) \cdot \hat{\boldsymbol{t}}\left|\frac{d \boldsymbol{x}}{d t}\right| d t=\int_{t_0}^{t_1} \boldsymbol{f}(\boldsymbol{x}(t)) \cdot \frac{d \boldsymbol{x}}{d t} d t .
$$
\end{definition}

\begin{nex}[Showing Line Integral depends on Curve]
	
	
	Take $\mathbf{F}(\mathbf{r})=\left(x e^y, z^2, x y\right)$ and we want to find the line integral from $\mathbf{a}=(0,0,0)$ to $\mathbf{b}=(1,1,1)$. 
	
	\begin{figure}[H]
		
		\centering
		\includegraphics[width=0.3\textwidth]{figures/lineintegral.pdf}
		\caption{Two curves between $a$ and $b$ }
	\end{figure}
	
	
	We first integrate along the curve $C_1: \mathbf{r}(u)=\left(u, u^2, u^3\right)$. Then $\mathbf{r}^{\prime}(u)= \left(1,2 u, 3 u^2\right)$, and $\mathbf{F}(\mathbf{r}(u))=\left(u e^{u^2}, u^6, u^3\right)$. So
	
	$$
	\begin{aligned}
		\int_{C_1} \mathbf{F} \cdot \mathrm{~d} \mathbf{r} & =\int_0^1 \mathbf{F} \cdot \mathbf{r}^{\prime}(u) \mathrm{d} u \\
		& =\int_0^1 u e^{u^2}+2 u^7+3 u^5 \mathrm{~d} u \\
		& =\frac{e}{2}-\frac{1}{2}+\frac{1}{4}+\frac{1}{2} \\
		& =\frac{e}{2}+\frac{1}{4}
	\end{aligned}
	$$
	
	Now we try to integrate along another curve $C_2: \mathbf{r}(t)=(t, t, t)$. So $\mathbf{r}^{\prime}(t)= (1,1,1)$.
	
	$$
	\begin{aligned}
		\int_{C_2} \mathbf{F} \cdot \mathrm{~d} \mathbf{r} & =\int \mathbf{F} \cdot \mathbf{r}^{\prime}(t) \mathrm{d} t \\
		& =\int_0^1 t e^t+2 t^2 \mathrm{~d} t \\
		& =\frac{5}{3}
	\end{aligned}
	$$
	
	
	We see that the line integral depends on the curve $C$ in general, not just $\mathbf{a}, \mathbf{b}$.
\end{nex}



\begin{remark}
  The parametrisation must trace the curve in the direction given by $\hat{\mathbf{t}}$
\end{remark}

\begin{remark}
  The standard example from physics here is the work done by a particle.
\end{remark}

\begin{definition}[Circulation of $\mathbf{f}$ around $C$]
  If the curve $C$ is closed, this is often emphasised by the notation

$$
\int_C \boldsymbol{f} \cdot \hat{\boldsymbol{t}} d \ell=\oint_C \boldsymbol{f} \cdot \hat{\boldsymbol{t}} d \ell
$$

and the line integral is called the circulation of $\boldsymbol{f}$ around $C$.
\end{definition}

\begin{ndefi}[Piecewise Smooth Curve]
	A piecewise smooth curve is a curve $C=C_1+C_2+\cdots+C_n$ with all $C_i$ smooth with regular parametrisations. The line integral over a piecewise smooth $C$ is
	
	$$
	\int_C \mathbf{F} \cdot \mathrm{~d} \mathbf{r}=\int_{C_1} \mathbf{F} \cdot \mathrm{~d} \mathbf{r}+\int_{C_2} \mathbf{F} \cdot \mathrm{~d} \mathbf{r}+\cdots+\int_{C_n} \mathbf{F} \cdot \mathrm{~d} \mathbf{r} .
	$$
\end{ndefi}

\begin{nex}[Turning two curves into piecewise smooth closed curve]
	
	
	Take the example above, and let $C_3=-C_2$. Then $C=C_1+C_3$ is piecewise smooth but not smooth:
	
	\begin{figure}[H]
		\centering
		\includegraphics[width=0.3\textwidth]{figures/closedlineintegral.pdf}
		\caption{Closed line integral between $a$ and $b$ }
	\end{figure}
	
	Then
	
	$$
	\begin{aligned}
		\oint_C \mathbf{F} \cdot \mathrm{~d} \mathbf{r} & =\int_{C_1} \mathbf{F} \cdot \mathrm{~d} \mathbf{r}+\int_{C_3} \mathbf{F} \cdot \mathrm{~d} \mathbf{r} \\
		& =\left(\frac{e}{2}+\frac{1}{4}\right)-\frac{5}{3} \\
		& =-\frac{17}{12}+\frac{e}{2}
	\end{aligned}
	$$
\end{nex}


\begin{remark}[Differential form for Line Integrals]
  You may also come across the "differential form" notation for line integrals,

$$
\int_C \boldsymbol{f} \cdot \hat{\boldsymbol{t}} d \ell=\int_C \boldsymbol{f} \cdot d \boldsymbol{x}
$$
\end{remark}

\section{Curvilinear Coordinates}

\begin{definition}[Curvilinear Coordinates]
  In a curvilinear coordinate system ( $u, v, w$ ), the coordinate lines - where two of the coordinates are constant and one varies - may be curved. These coordinate lines may be parametrised by the corresponding coordinate, so are given by $\boldsymbol{x}(u, v, w)$ with two of the coordinates fixed.
\end{definition}

\begin{definition}[Tangent Vectors and Unit Tangents]

  For a system $\mathbf{x}(u,v,w)$ we have the corresponding tangent vectors:

  $$\frac{\partial \boldsymbol{x}}{\partial u}, \frac{\partial \boldsymbol{x}}{\partial v}, \frac{\partial \boldsymbol{x}}{\partial w}$$.

\noindent From these tangent vectors comes the unit tangent vectors:

$$
\boldsymbol{e}_u=\frac{1}{h_u} \frac{\partial \boldsymbol{x}}{\partial u}, \quad \boldsymbol{e}_v=\frac{1}{h_v} \frac{\partial \boldsymbol{x}}{\partial v}, \quad \boldsymbol{e}_w=\frac{1}{h_w} \frac{\partial \boldsymbol{x}}{\partial w},
$$

\noindent where

$$
h_u=\left|\frac{\partial \boldsymbol{x}}{\partial u}\right|, \quad h_v=\left|\frac{\partial \boldsymbol{x}}{\partial v}\right|, \quad h_w=\left|\frac{\partial \boldsymbol{x}}{\partial w}\right|
$$

\noindent are called the scale factors.
\end{definition}

\begin{remark}
  A set of tangent vectors form an orthonormal basis if the all of them are mutually orthogonal.
\end{remark}

\begin{remark} The scale factors tell us how much an infinitesimal region in $(u,v,w)$ coordinates is stretched at different points in real space. They are crucial when calculating surface and volume integrals. We do not require systems to be orthogonal when we take their coordinates.
\end{remark}

\begin{definition}[Cylindrical and Spherical Coordinates]

  
  \begin{itemize} \emph{}

    \item[1.] \textbf{Cylindrical coordinates} are useful for problems with axial symmetry:

$$
\boldsymbol{x}(r, \theta, z)=r \cos \theta \boldsymbol{e}_1+r \sin \theta \boldsymbol{e}_2+z \boldsymbol{e}_3, \quad r>0, \quad \theta \in[0,2 \pi], \quad z \in \mathbb{R},
$$

leading to scale factors $h_r=1, h_\theta=r, h_z=1$.
\item[2.] \textbf{Spherical coordinates} are useful for problems with spherical symmetry:

$$
\boldsymbol{x}(r, \theta, \phi)=r \sin \theta \cos \phi \boldsymbol{e}_1+r \sin \theta \sin \phi \boldsymbol{e}_2+r \cos \theta \boldsymbol{e}_3, \quad r>0, \quad \theta \in[0, \pi], \quad \phi \in[0,2 \pi],
$$

leading to scale factors $h_r=1, h_\theta=r, h_\phi=r \sin \theta$.
\end{itemize}
  
\end{definition}
